## 1. First Unique Character

### Problem Understanding:
Find the first non-repeating character in a string and return its index. Return -1 if none exists.

**Example:**
```
Input: "leetcode"
Output: 0 (l is first unique character)

Input: "loveleetcode"
Output: 2 (v is first unique character)
```

### Solutions:

```javascript
// Method 1: Two Pass with Frequency Count (Optimal)
function firstUniqChar(s) {
    const frequency = new Array(26).fill(0); // For lowercase English letters
    
    // First pass: count frequency of each character
    for (let i = 0; i < s.length; i++) {
        const charCode = s.charCodeAt(i) - 97; // 'a' = 97
        frequency[charCode]++;
    }
    
    // Second pass: find first character with frequency 1
    for (let i = 0; i < s.length; i++) {
        const charCode = s.charCodeAt(i) - 97;
        if (frequency[charCode] === 1) {
            return i;
        }
    }
    
    return -1;
}

// Method 2: Using Map for General Characters
function firstUniqCharMap(s) {
    const frequency = new Map();
    
    // First pass: build frequency map
    for (let i = 0; i < s.length; i++) {
        const char = s[i];
        frequency.set(char, (frequency.get(char) || 0) + 1);
    }
    
    // Second pass: find first unique character
    for (let i = 0; i < s.length; i++) {
        if (frequency.get(s[i]) === 1) {
            return i;
        }
    }
    
    return -1;
}

// Method 3: Using Object (Similar to Map)
function firstUniqCharObject(s) {
    const frequency = {};
    
    // Count frequencies
    for (let char of s) {
        frequency[char] = (frequency[char] || 0) + 1;
    }
    
    // Find first unique
    for (let i = 0; i < s.length; i++) {
        if (frequency[s[i]] === 1) {
            return i;
        }
    }
    
    return -1;
}

// Method 4: One Pass with Linked List (Advanced)
class Node {
    constructor(char, index) {
        this.char = char;
        this.index = index;
        this.prev = null;
        this.next = null;
    }
}

function firstUniqCharLinkedList(s) {
    const map = new Map(); // char -> Node
    const head = new Node(null, -1);
    const tail = new Node(null, -1);
    head.next = tail;
    tail.prev = head;
    
    function addNode(node) {
        const prev = tail.prev;
        prev.next = node;
        node.prev = prev;
        node.next = tail;
        tail.prev = node;
    }
    
    function removeNode(node) {
        const prev = node.prev;
        const next = node.next;
        prev.next = next;
        next.prev = prev;
    }
    
    for (let i = 0; i < s.length; i++) {
        const char = s[i];
        
        if (map.has(char)) {
            // Character seen before, remove from list
            const node = map.get(char);
            if (node) {
                removeNode(node);
                map.set(char, null); // Mark as duplicate
            }
        } else {
            // First time seeing character
            const node = new Node(char, i);
            addNode(node);
            map.set(char, node);
        }
    }
    
    return head.next !== tail ? head.next.index : -1;
}

// Enhanced version with visualization
function firstUniqCharWithSteps(s) {
    const steps = [];
    const frequency = new Map();
    const firstOccurrence = new Map();
    
    steps.push(`Input string: "${s}"`);
    
    for (let i = 0; i < s.length; i++) {
        const char = s[i];
        const currentFreq = (frequency.get(char) || 0) + 1;
        frequency.set(char, currentFreq);
        
        if (currentFreq === 1) {
            firstOccurrence.set(char, i);
            steps.push(`Character '${char}' at index ${i}: first occurrence`);
        } else {
            steps.push(`Character '${char}' at index ${i}: duplicate (frequency ${currentFreq})`);
        }
    }
    
    let result = -1;
    for (let [char, freq] of frequency) {
        if (freq === 1) {
            const index = firstOccurrence.get(char);
            steps.push(`Found unique character '${char}' at index ${index}`);
            if (result === -1 || index < result) {
                result = index;
            }
        }
    }
    
    steps.push(result === -1 ? "No unique character found" : `First unique character at index ${result}`);
    
    return { result, steps };
}
```

## 2. Group Anagrams

### Problem Understanding:
Group strings that are anagrams of each other. Anagrams are words formed by rearranging letters.

**Example:**
```
Input: ["eat","tea","tan","ate","nat","bat"]
Output: [["bat"],["nat","tan"],["ate","eat","tea"]]
```

### Solutions:

```javascript
// Method 1: Sort + Hash Map (Most Common)
function groupAnagrams(strs) {
    const groups = new Map();
    
    for (let str of strs) {
        // Sort characters to get canonical form
        const sorted = str.split('').sort().join('');
        
        if (!groups.has(sorted)) {
            groups.set(sorted, []);
        }
        groups.get(sorted).push(str);
    }
    
    return Array.from(groups.values());
}

// Method 2: Character Count as Key (More Efficient)
function groupAnagramsCount(strs) {
    const groups = new Map();
    
    for (let str of strs) {
        const count = new Array(26).fill(0);
        
        // Count frequency of each character
        for (let char of str) {
            const index = char.charCodeAt(0) - 97; // 'a' = 97
            count[index]++;
        }
        
        // Use count array as key (convert to string)
        const key = count.join('#');
        
        if (!groups.has(key)) {
            groups.set(key, []);
        }
        groups.get(key).push(str);
    }
    
    return Array.from(groups.values());
}

// Method 3: Using Prime Number Multiplication
function groupAnagramsPrime(strs) {
    // Assign prime numbers to each letter
    const primes = {
        'a': 2, 'b': 3, 'c': 5, 'd': 7, 'e': 11, 'f': 13, 'g': 17, 'h': 19,
        'i': 23, 'j': 29, 'k': 31, 'l': 37, 'm': 41, 'n': 43, 'o': 47,
        'p': 53, 'q': 59, 'r': 61, 's': 67, 't': 71, 'u': 73, 'v': 79,
        'w': 83, 'x': 89, 'y': 97, 'z': 101
    };
    
    const groups = new Map();
    
    for (let str of strs) {
        let product = 1;
        
        // Multiply prime numbers for each character
        for (let char of str) {
            product *= primes[char];
        }
        
        if (!groups.has(product)) {
            groups.set(product, []);
        }
        groups.get(product).push(str);
    }
    
    return Array.from(groups.values());
}

// Method 4: Using Object with Sorted Keys
function groupAnagramsObject(strs) {
    const groups = {};
    
    for (let str of strs) {
        const key = str.split('').sort().join('');
        
        if (!groups[key]) {
            groups[key] = [];
        }
        groups[key].push(str);
    }
    
    return Object.values(groups);
}

// Enhanced version with visualization
function groupAnagramsWithSteps(strs) {
    const steps = [];
    const groups = new Map();
    
    steps.push(`Input: [${strs}]`);
    
    for (let i = 0; i < strs.length; i++) {
        const str = strs[i];
        const sorted = str.split('').sort().join('');
        
        steps.push(`Processing "${str}" -> sorted: "${sorted}"`);
        
        if (!groups.has(sorted)) {
            groups.set(sorted, []);
            steps.push(`  Created new group for key "${sorted}"`);
        }
        
        groups.get(sorted).push(str);
        steps.push(`  Added to group: [${groups.get(sorted)}]`);
    }
    
    const result = Array.from(groups.values());
    steps.push(`Final groups: ${JSON.stringify(result)}`);
    
    return { result, steps };
}
```

**Space Complexity:** O(n × k) for storing all strings

## 3. Top K Frequent Elements

### Problem Understanding:
Given an array of integers, return the k most frequent elements.

**Example:**
```
Input: nums = [1,1,1,2,2,3], k = 2
Output: [1,2]
```

### Solutions:

```javascript
// Method 1: Bucket Sort (Optimal for this problem)
function topKFrequent(nums, k) {
    const frequency = new Map();
    
    // Count frequencies
    for (let num of nums) {
        frequency.set(num, (frequency.get(num) || 0) + 1);
    }
    
    // Create buckets - index represents frequency
    const buckets = new Array(nums.length + 1);
    for (let i = 0; i < buckets.length; i++) {
        buckets[i] = [];
    }
    
    // Place numbers in buckets based on frequency
    for (let [num, freq] of frequency) {
        buckets[freq].push(num);
    }
    
    // Collect top k frequent elements
    const result = [];
    for (let i = buckets.length - 1; i >= 0 && result.length < k; i--) {
        if (buckets[i].length > 0) {
            result.push(...buckets[i]);
        }
    }
    
    return result.slice(0, k);
}

// Method 2: Min Heap (Priority Queue)
class MinHeap {
    constructor() {
        this.heap = [];
    }
    
    push(val, priority) {
        this.heap.push({ val, priority });
        this._bubbleUp();
    }
    
    pop() {
        const min = this.heap[0];
        const end = this.heap.pop();
        
        if (this.heap.length > 0) {
            this.heap[0] = end;
            this._sinkDown();
        }
        
        return min;
    }
    
    size() {
        return this.heap.length;
    }
    
    _bubbleUp() {
        let index = this.heap.length - 1;
        const element = this.heap[index];
        
        while (index > 0) {
            const parentIndex = Math.floor((index - 1) / 2);
            const parent = this.heap[parentIndex];
            
            if (element.priority >= parent.priority) break;
            
            this.heap[parentIndex] = element;
            this.heap[index] = parent;
            index = parentIndex;
        }
    }
    
    _sinkDown() {
        let index = 0;
        const length = this.heap.length;
        const element = this.heap[0];
        
        while (true) {
            const leftChildIndex = 2 * index + 1;
            const rightChildIndex = 2 * index + 2;
            let leftChild, rightChild;
            let swap = null;
            
            if (leftChildIndex < length) {
                leftChild = this.heap[leftChildIndex];
                if (leftChild.priority < element.priority) {
                    swap = leftChildIndex;
                }
            }
            
            if (rightChildIndex < length) {
                rightChild = this.heap[rightChildIndex];
                if ((swap === null && rightChild.priority < element.priority) ||
                    (swap !== null && rightChild.priority < leftChild.priority)) {
                    swap = rightChildIndex;
                }
            }
            
            if (swap === null) break;
            
            this.heap[index] = this.heap[swap];
            this.heap[swap] = element;
            index = swap;
        }
    }
}

function topKFrequentHeap(nums, k) {
    const frequency = new Map();
    
    // Count frequencies
    for (let num of nums) {
        frequency.set(num, (frequency.get(num) || 0) + 1);
    }
    
    // Use min heap to keep track of top k elements
    const heap = new MinHeap();
    
    for (let [num, freq] of frequency) {
        heap.push(num, freq);
        
        // If heap size exceeds k, remove the least frequent
        if (heap.size() > k) {
            heap.pop();
        }
    }
    
    // Extract elements from heap
    const result = [];
    while (heap.size() > 0) {
        result.push(heap.pop().val);
    }
    
    return result.reverse(); // Most frequent first
}

// Method 3: QuickSelect (Optimized)
function topKFrequentQuickSelect(nums, k) {
    const frequency = new Map();
    
    for (let num of nums) {
        frequency.set(num, (frequency.get(num) || 0) + 1);
    }
    
    const unique = Array.from(frequency.keys());
    
    function quickSelect(left, right, k) {
        const pivotIndex = partition(left, right);
        
        if (pivotIndex === k) {
            return;
        } else if (pivotIndex < k) {
            quickSelect(pivotIndex + 1, right, k);
        } else {
            quickSelect(left, pivotIndex - 1, k);
        }
    }
    
    function partition(left, right) {
        const pivotFreq = frequency.get(unique[right]);
        let i = left;
        
        for (let j = left; j < right; j++) {
            if (frequency.get(unique[j]) >= pivotFreq) {
                [unique[i], unique[j]] = [unique[j], unique[i]];
                i++;
            }
        }
        
        [unique[i], unique[right]] = [unique[right], unique[i]];
        return i;
    }
    
    quickSelect(0, unique.length - 1, k - 1);
    return unique.slice(0, k);
}

// Enhanced version with visualization
function topKFrequentWithSteps(nums, k) {
    const steps = [];
    const frequency = new Map();
    
    steps.push(`Input: [${nums}], k = ${k}`);
    
    // Count frequencies
    for (let num of nums) {
        const currentFreq = (frequency.get(num) || 0) + 1;
        frequency.set(num, currentFreq);
        steps.push(`Number ${num}: frequency = ${currentFreq}`);
    }
    
    steps.push(`Frequency map: ${JSON.stringify(Array.from(frequency))}`);
    
    // Bucket sort approach
    const buckets = new Array(nums.length + 1);
    for (let i = 0; i < buckets.length; i++) {
        buckets[i] = [];
    }
    
    for (let [num, freq] of frequency) {
        buckets[freq].push(num);
        steps.push(`Placed ${num} (freq=${freq}) in bucket ${freq}`);
    }
    
    const result = [];
    for (let i = buckets.length - 1; i >= 0 && result.length < k; i--) {
        if (buckets[i].length > 0) {
            steps.push(`Bucket ${i}: [${buckets[i]}] - adding to result`);
            result.push(...buckets[i]);
        }
    }
    
    const finalResult = result.slice(0, k);
    steps.push(`Final result: [${finalResult}]`);
    
    return { result: finalResult, steps };
}

```

**Space Complexity:** O(n) for all methods

## 4. Find Median from Data Stream

### Problem Understanding:
Design a data structure that supports adding numbers and finding the median of all numbers efficiently.

**Example:**
```
addNum(1)
addNum(2)
findMedian() -> 1.5
addNum(3) 
findMedian() -> 2
```

### Solutions:

```javascript
// Method 1: Two Heaps (Optimal)
class MedianFinder {
    constructor() {
        // Max heap for lower half (implemented using min heap with negative values)
        this.lower = new MinHeap();
        // Min heap for upper half
        this.upper = new MinHeap();
    }
    
    addNum(num) {
        // Add to appropriate heap
        if (this.lower.size() === 0 || num <= -this.lower.heap[0].priority) {
            this.lower.push(num, -num); // Store as negative for max heap behavior
        } else {
            this.upper.push(num, num);
        }
        
        // Balance heaps
        this._balanceHeaps();
    }
    
    findMedian() {
        if (this.lower.size() > this.upper.size()) {
            return -this.lower.heap[0].priority;
        } else {
            return (-this.lower.heap[0].priority + this.upper.heap[0].priority) / 2;
        }
    }
    
    _balanceHeaps() {
        // Ensure lower heap has at most one more element than upper heap
        if (this.lower.size() > this.upper.size() + 1) {
            const moved = this.lower.pop();
            this.upper.push(moved.val, -moved.priority);
        } else if (this.upper.size() > this.lower.size()) {
            const moved = this.upper.pop();
            this.lower.push(moved.val, -moved.val);
        }
    }
}

// Method 2: Using Built-in Sort (Naive)
class MedianFinderNaive {
    constructor() {
        this.numbers = [];
    }
    
    addNum(num) {
        this.numbers.push(num);
        this.numbers.sort((a, b) => a - b);
    }
    
    findMedian() {
        const n = this.numbers.length;
        if (n % 2 === 0) {
            return (this.numbers[n/2 - 1] + this.numbers[n/2]) / 2;
        } else {
            return this.numbers[Math.floor(n/2)];
        }
    }
}

// Method 3: Insertion Sort Style
class MedianFinderInsertion {
    constructor() {
        this.numbers = [];
    }
    
    addNum(num) {
        // Binary search to find insertion position
        let left = 0, right = this.numbers.length - 1;
        while (left <= right) {
            const mid = Math.floor((left + right) / 2);
            if (this.numbers[mid] < num) {
                left = mid + 1;
            } else {
                right = mid - 1;
            }
        }
        
        // Insert at found position
        this.numbers.splice(left, 0, num);
    }
    
    findMedian() {
        const n = this.numbers.length;
        if (n % 2 === 0) {
            return (this.numbers[n/2 - 1] + this.numbers[n/2]) / 2;
        } else {
            return this.numbers[Math.floor(n/2)];
        }
    }
}

// Enhanced MinHeap for median finder
class EnhancedMinHeap {
    constructor() {
        this.heap = [];
    }
    
    push(val, priority) {
        this.heap.push({ val, priority });
        this._bubbleUp(this.heap.length - 1);
    }
    
    pop() {
        const min = this.heap[0];
        const last = this.heap.pop();
        
        if (this.heap.length > 0) {
            this.heap[0] = last;
            this._sinkDown(0);
        }
        
        return min;
    }
    
    peek() {
        return this.heap[0];
    }
    
    size() {
        return this.heap.length;
    }
    
    _bubbleUp(index) {
        const element = this.heap[index];
        
        while (index > 0) {
            const parentIndex = Math.floor((index - 1) / 2);
            const parent = this.heap[parentIndex];
            
            if (element.priority >= parent.priority) break;
            
            this.heap[parentIndex] = element;
            this.heap[index] = parent;
            index = parentIndex;
        }
    }
    
    _sinkDown(index) {
        const length = this.heap.length;
        const element = this.heap[index];
        
        while (true) {
            let leftChildIndex = 2 * index + 1;
            let rightChildIndex = 2 * index + 2;
            let leftChild, rightChild;
            let swap = null;
            
            if (leftChildIndex < length) {
                leftChild = this.heap[leftChildIndex];
                if (leftChild.priority < element.priority) {
                    swap = leftChildIndex;
                }
            }
            
            if (rightChildIndex < length) {
                rightChild = this.heap[rightChildIndex];
                if ((swap === null && rightChild.priority < element.priority) ||
                    (swap !== null && rightChild.priority < leftChild.priority)) {
                    swap = rightChildIndex;
                }
            }
            
            if (swap === null) break;
            
            this.heap[index] = this.heap[swap];
            this.heap[swap] = element;
            index = swap;
        }
    }
}

class TrackingMedianFinder extends MedianFinder {

        this.operations = [];

    }
```

## 🚀 Key Patterns Learned:

1. **Frequency Counting**: Hash maps for character/word counts
