## 1. Climbing Stairs

### Problem Understanding:
You are climbing a staircase. It takes `n` steps to reach the top. Each time you can either climb 1 or 2 steps. In how many distinct ways can you climb to the top?

**Example:**
```
Input: n = 3
Output: 3
Explanation: 
1. 1 step + 1 step + 1 step
2. 1 step + 2 steps
3. 2 steps + 1 step
```

### Solutions:

```javascript
// Method 1: Recursive (Top-down) - Time: O(2^n), Space: O(n)
function climbStairsRecursive(n) {
    if (n <= 2) return n;
    return climbStairsRecursive(n - 1) + climbStairsRecursive(n - 2);
}

// Method 2: Memoization (Top-down DP) - Time: O(n), Space: O(n)
function climbStairsMemo(n) {
    const memo = new Array(n + 1).fill(-1);
    
    function dp(i) {
        if (i <= 2) return i;
        if (memo[i] !== -1) return memo[i];
        
        memo[i] = dp(i - 1) + dp(i - 2);
        return memo[i];
    }
    
    return dp(n);
}

// Method 3: Tabulation (Bottom-up DP) - Time: O(n), Space: O(n)
function climbStairsTabulation(n) {
    if (n <= 2) return n;
    
    const dp = new Array(n + 1);
    dp[1] = 1;
    dp[2] = 2;
    
    for (let i = 3; i <= n; i++) {
        dp[i] = dp[i - 1] + dp[i - 2];
    }
    
    return dp[n];
}

// Method 4: Space Optimized - Time: O(n), Space: O(1)
function climbStairsOptimized(n) {
    if (n <= 2) return n;
    
    let prev1 = 1; // ways for n-1
    let prev2 = 2; // ways for n-2
    
    for (let i = 3; i <= n; i++) {
        const current = prev1 + prev2;
        prev1 = prev2;
        prev2 = current;
    }
    
    return prev2;
}

// Method 5: Matrix Exponentiation (O(log n) time) - Advanced
function climbStairsMatrix(n) {
    if (n <= 2) return n;
    
    function multiply(a, b) {
        return [
            a[0] * b[0] + a[1] * b[2],
            a[0] * b[1] + a[1] * b[3],
            a[2] * b[0] + a[3] * b[2],
            a[2] * b[1] + a[3] * b[3]
        ];
    }
    
    function power(matrix, n) {
        let result = [1, 0, 0, 1]; // Identity matrix
        let base = matrix;
        
        while (n > 0) {
            if (n % 2 === 1) {
                result = multiply(result, base);
            }
            base = multiply(base, base);
            n = Math.floor(n / 2);
        }
        
        return result;
    }
    
    const baseMatrix = [1, 1, 1, 0];
    const resultMatrix = power(baseMatrix, n);
    return resultMatrix[1]; // or resultMatrix[2]
}

// Enhanced version with visualization
function climbStairsWithSteps(n) {
    const steps = [];
    
    if (n <= 2) {
        steps.push(`Base case: n = ${n}, ways = ${n}`);
        return { ways: n, steps };
    }
    
    const dp = new Array(n + 1);
    dp[1] = 1;
    dp[2] = 2;
    steps.push(`dp[1] = 1 (only: 1)`);
    steps.push(`dp[2] = 2 (ways: 1+1, 2)`);
    
    for (let i = 3; i <= n; i++) {
        dp[i] = dp[i - 1] + dp[i - 2];
        steps.push(`dp[${i}] = dp[${i-1}] + dp[${i-2}] = ${dp[i-1]} + ${dp[i-2]} = ${dp[i]}`);
        steps.push(`  From step ${i}: take 1 step (from ${i-1}) or 2 steps (from ${i-2})`);
    }
    
    return { ways: dp[n], steps };
}
```

**Space Complexity:** O(n) for memo/tabulation, O(1) for optimized

## 2. Coin Change

### Problem Understanding:
Given coins of different denominations and a total amount, find the minimum number of coins needed to make that amount. Return -1 if it's not possible.

**Example:**
```
Input: coins = [1,2,5], amount = 11
Output: 3 (5 + 5 + 1)
```

### Solutions:

```javascript
// Method 1: Recursive (Brute Force) - Time: O(S^n), Space: O(n)
function coinChangeRecursive(coins, amount) {
    function dp(remaining) {
        if (remaining === 0) return 0;
        if (remaining < 0) return Infinity;
        
        let minCoins = Infinity;
        for (let coin of coins) {
            const result = dp(remaining - coin);
            if (result !== Infinity) {
                minCoins = Math.min(minCoins, result + 1);
            }
        }
        
        return minCoins;
    }
    
    const result = dp(amount);
    return result === Infinity ? -1 : result;
}

// Method 2: Memoization (Top-down DP) - Time: O(S × n), Space: O(S)
function coinChangeMemo(coins, amount) {
    const memo = new Map();
    
    function dp(remaining) {
        if (remaining === 0) return 0;
        if (remaining < 0) return Infinity;
        if (memo.has(remaining)) return memo.get(remaining);
        
        let minCoins = Infinity;
        for (let coin of coins) {
            const result = dp(remaining - coin);
            if (result !== Infinity) {
                minCoins = Math.min(minCoins, result + 1);
            }
        }
        
        memo.set(remaining, minCoins);
        return minCoins;
    }
    
    const result = dp(amount);
    return result === Infinity ? -1 : result;
}

// Method 3: Tabulation (Bottom-up DP) - Time: O(S × n), Space: O(S)
function coinChange(coins, amount) {
    // dp[i] represents min coins needed for amount i
    const dp = new Array(amount + 1).fill(Infinity);
    dp[0] = 0; // Base case: 0 coins needed for amount 0
    
    for (let i = 1; i <= amount; i++) {
        for (let coin of coins) {
            if (i - coin >= 0) {
                dp[i] = Math.min(dp[i], dp[i - coin] + 1);
            }
        }
    }
    
    return dp[amount] === Infinity ? -1 : dp[amount];
}

// Method 4: BFS Approach (for minimum coins)
function coinChangeBFS(coins, amount) {
    if (amount === 0) return 0;
    
    const queue = [0]; // amounts
    const visited = new Set([0]);
    let level = 0;
    
    while (queue.length > 0) {
        const levelSize = queue.length;
        level++;
        
        for (let i = 0; i < levelSize; i++) {
            const current = queue.shift();
            
            for (let coin of coins) {
                const nextAmount = current + coin;
                
                if (nextAmount === amount) {
                    return level;
                }
                
                if (nextAmount < amount && !visited.has(nextAmount)) {
                    visited.add(nextAmount);
                    queue.push(nextAmount);
                }
            }
        }
    }
    
    return -1;
}

// Method 5: Coin Change II (Number of combinations)
function coinChangeII(amount, coins) {
    // dp[i] represents number of combinations for amount i
    const dp = new Array(amount + 1).fill(0);
    dp[0] = 1; // 1 way to make amount 0 (use no coins)
    
    for (let coin of coins) {
        for (let i = coin; i <= amount; i++) {
            dp[i] += dp[i - coin];
        }
    }
    
    return dp[amount];
}

// Enhanced version with visualization
function coinChangeWithSteps(coins, amount) {
    const steps = [];
    
    const dp = new Array(amount + 1).fill(Infinity);
    dp[0] = 0;
    steps.push(`dp[0] = 0 (base case: 0 coins for amount 0)`);
    
    for (let i = 1; i <= amount; i++) {
        steps.push(`\nCalculating dp[${i}]:`);
        let minCoins = Infinity;
        
        for (let coin of coins) {
            if (i - coin >= 0) {
                const candidate = dp[i - coin] + 1;
                steps.push(`  Try coin ${coin}: dp[${i}-${coin}] + 1 = dp[${i-coin}] + 1 = ${candidate}`);
                minCoins = Math.min(minCoins, candidate);
            } else {
                steps.push(`  Try coin ${coin}: not possible (${i}-${coin} < 0)`);
            }
        }
        
        dp[i] = minCoins;
        steps.push(`  dp[${i}] = ${minCoins === Infinity ? 'Infinity' : minCoins}`);
    }
    
    const result = dp[amount] === Infinity ? -1 : dp[amount];
    steps.push(`\nFinal result: ${result}`);
    
    if (result !== -1) {
        // Reconstruct solution
        const solution = [];
        let remaining = amount;
        while (remaining > 0) {
            for (let coin of coins) {
                if (remaining - coin >= 0 && dp[remaining] === dp[remaining - coin] + 1) {
                    solution.push(coin);
                    remaining -= coin;
                    break;
                }
            }
        }
        steps.push(`Coin combination: [${solution.join(', ')}]`);
    }
    
    return { minCoins: result, steps };
}

```

**Space Complexity:** O(S) for DP array

## 3. Longest Common Subsequence

### Problem Understanding:
Given two strings, find the length of their longest common subsequence (LCS). A subsequence is a sequence that appears in the same relative order, but not necessarily contiguous.

**Example:**
```
Input: text1 = "abcde", text2 = "ace" 
Output: 3 (LCS is "ace")
```

### Solutions:

```javascript
// Method 1: Recursive (Brute Force) - Time: O(2^(m+n)), Space: O(m+n)
function longestCommonSubsequenceRecursive(text1, text2) {
    function dp(i, j) {
        if (i === text1.length || j === text2.length) return 0;
        
        if (text1[i] === text2[j]) {
            return 1 + dp(i + 1, j + 1);
        } else {
            return Math.max(dp(i + 1, j), dp(i, j + 1));
        }
    }
    
    return dp(0, 0);
}

// Method 2: Memoization (Top-down DP) - Time: O(m × n), Space: O(m × n)
function longestCommonSubsequenceMemo(text1, text2) {
    const m = text1.length, n = text2.length;
    const memo = new Array(m).fill(0).map(() => new Array(n).fill(-1));
    
    function dp(i, j) {
        if (i === m || j === n) return 0;
        if (memo[i][j] !== -1) return memo[i][j];
        
        if (text1[i] === text2[j]) {
            memo[i][j] = 1 + dp(i + 1, j + 1);
        } else {
            memo[i][j] = Math.max(dp(i + 1, j), dp(i, j + 1));
        }
        
        return memo[i][j];
    }
    
    return dp(0, 0);
}

// Method 3: Tabulation (Bottom-up DP) - Time: O(m × n), Space: O(m × n)
function longestCommonSubsequence(text1, text2) {
    const m = text1.length, n = text2.length;
    const dp = new Array(m + 1).fill(0).map(() => new Array(n + 1).fill(0));
    
    for (let i = 1; i <= m; i++) {
        for (let j = 1; j <= n; j++) {
            if (text1[i - 1] === text2[j - 1]) {
                dp[i][j] = 1 + dp[i - 1][j - 1];
            } else {
                dp[i][j] = Math.max(dp[i - 1][j], dp[i][j - 1]);
            }
        }
    }
    
    return dp[m][n];
}

// Method 4: Space Optimized - Time: O(m × n), Space: O(min(m, n))
function longestCommonSubsequenceOptimized(text1, text2) {
    // Ensure text1 is the shorter string for space optimization
    if (text1.length > text2.length) {
        [text1, text2] = [text2, text1];
    }
    
    const m = text1.length, n = text2.length;
    let prev = new Array(m + 1).fill(0);
    let curr = new Array(m + 1).fill(0);
    
    for (let j = 1; j <= n; j++) {
        for (let i = 1; i <= m; i++) {
            if (text1[i - 1] === text2[j - 1]) {
                curr[i] = 1 + prev[i - 1];
            } else {
                curr[i] = Math.max(prev[i], curr[i - 1]);
            }
        }
        [prev, curr] = [curr, prev];
    }
    
    return prev[m];
}

// Method 5: Reconstruct LCS String
function longestCommonSubsequenceString(text1, text2) {
    const m = text1.length, n = text2.length;
    const dp = new Array(m + 1).fill(0).map(() => new Array(n + 1).fill(0));
    
    // Build DP table
    for (let i = 1; i <= m; i++) {
        for (let j = 1; j <= n; j++) {
            if (text1[i - 1] === text2[j - 1]) {
                dp[i][j] = 1 + dp[i - 1][j - 1];
            } else {
                dp[i][j] = Math.max(dp[i - 1][j], dp[i][j - 1]);
            }
        }
    }
    
    // Reconstruct LCS
    let lcs = '';
    let i = m, j = n;
    
    while (i > 0 && j > 0) {
        if (text1[i - 1] === text2[j - 1]) {
            lcs = text1[i - 1] + lcs;
            i--;
            j--;
        } else if (dp[i - 1][j] > dp[i][j - 1]) {
            i--;
        } else {
            j--;
        }
    }
    
    return lcs;
}

// Enhanced version with visualization
function longestCommonSubsequenceWithSteps(text1, text2) {
    const steps = [];
    const m = text1.length, n = text2.length;
    const dp = new Array(m + 1).fill(0).map(() => new Array(n + 1).fill(0));
    
    steps.push(`Text1: "${text1}" (length ${m})`);
    steps.push(`Text2: "${text2}" (length ${n})`);
    steps.push('\nBuilding DP table:');
    
    for (let i = 1; i <= m; i++) {
        for (let j = 1; j <= n; j++) {
            if (text1[i - 1] === text2[j - 1]) {
                dp[i][j] = 1 + dp[i - 1][j - 1];
                steps.push(`  dp[${i}][${j}] = 1 + dp[${i-1}][${j-1}] = ${dp[i][j]} (match: '${text1[i-1]}')`);
            } else {
                dp[i][j] = Math.max(dp[i - 1][j], dp[i][j - 1]);
                steps.push(`  dp[${i}][${j}] = max(dp[${i-1}][${j}]=${dp[i-1][j]}, dp[${i}][${j-1}]=${dp[i][j-1]}) = ${dp[i][j]}`);
            }
        }
    }
    
    const lcsLength = dp[m][n];
    const lcsString = reconstructLCS(text1, text2, dp);
    
    steps.push(`\nLCS Length: ${lcsLength}`);
    steps.push(`LCS String: "${lcsString}"`);
    
    return { length: lcsLength, string: lcsString, steps };
}

function reconstructLCS(text1, text2, dp) {
    let lcs = '';
    let i = text1.length, j = text2.length;
    
    while (i > 0 && j > 0) {
        if (text1[i - 1] === text2[j - 1]) {
            lcs = text1[i - 1] + lcs;
            i--;
            j--;
        } else if (dp[i - 1][j] > dp[i][j - 1]) {
            i--;
        } else {
            j--;
        }
    }
    
    return lcs;
}

console.log('LCS("abc", "abc"):', longestCommonSubsequence("abc", "abc")); // 3
console.log('LCS("abc", "def"):', longestCommonSubsequence("abc", "def")); // 0

console.log('\nLCS String:', longestCommonSubsequenceString("abcde", "ace")); // "ace"

// Test with steps
const withSteps = longestCommonSubsequenceWithSteps("abcde", "ace");
console.log("\nLCS with steps (first 10 steps):");
console.log(withSteps.steps.slice(0, 10));
```

**Time Complexity:** O(m × n)  
**Space Complexity:** O(m × n) for full table, O(min(m, n)) for optimized

## 4. House Robber

### Problem Understanding:
You are a robber planning to rob houses along a street. Each house has a certain amount of money. You cannot rob two adjacent houses. Determine the maximum amount you can rob.

**Example:**
```
Input: [2,7,9,3,1]
Output: 12 (2 + 9 + 1)
```

### Solutions:

```javascript
// Method 1: Recursive (Brute Force) - Time: O(2^n), Space: O(n)
function robRecursive(nums) {
    function dp(i) {
        if (i < 0) return 0;
        return Math.max(dp(i - 1), nums[i] + dp(i - 2));
    }
    
    return dp(nums.length - 1);
}

// Method 2: Memoization (Top-down DP) - Time: O(n), Space: O(n)
function robMemo(nums) {
    const memo = new Map();
    
    function dp(i) {
        if (i < 0) return 0;
        if (memo.has(i)) return memo.get(i);
        
        const result = Math.max(dp(i - 1), nums[i] + dp(i - 2));
        memo.set(i, result);
        return result;
    }
    
    return dp(nums.length - 1);
}

// Method 3: Tabulation (Bottom-up DP) - Time: O(n), Space: O(n)
function rob(nums) {
    if (nums.length === 0) return 0;
    if (nums.length === 1) return nums[0];
    
    const dp = new Array(nums.length);
    dp[0] = nums[0];
    dp[1] = Math.max(nums[0], nums[1]);
    
    for (let i = 2; i < nums.length; i++) {
        dp[i] = Math.max(dp[i - 1], nums[i] + dp[i - 2]);
    }
    
    return dp[nums.length - 1];
}

// Method 4: Space Optimized - Time: O(n), Space: O(1)
function robOptimized(nums) {
    if (nums.length === 0) return 0;
    
    let prev1 = 0; // dp[i-1]
    let prev2 = 0; // dp[i-2]
    
    for (let num of nums) {
        const current = Math.max(prev1, num + prev2);
        prev2 = prev1;
        prev1 = current;
    }
    
    return prev1;
}

// Method 5: House Robber II (Circular Street)
function robCircular(nums) {
    if (nums.length === 0) return 0;
    if (nums.length === 1) return nums[0];
    
    // Two cases: rob first house or rob last house
    function robLinear(houses) {
        let prev1 = 0, prev2 = 0;
        for (let num of houses) {
            const current = Math.max(prev1, num + prev2);
            prev2 = prev1;
            prev1 = current;
        }
        return prev1;
    }
    
    return Math.max(
        robLinear(nums.slice(0, nums.length - 1)), // exclude last house
        robLinear(nums.slice(1))                   // exclude first house
    );
}

// Method 6: House Robber III (Binary Tree)
function robTree(root) {
    function dfs(node) {
        if (!node) return [0, 0]; // [rob this node, skip this node]
        
        const left = dfs(node.left);
        const right = dfs(node.right);
        
        // If we rob this node, we cannot rob its children
        const rob = node.val + left[1] + right[1];
        
        // If we skip this node, we can rob its children (choose max)
        const skip = Math.max(left[0], left[1]) + Math.max(right[0], right[1]);
        
        return [rob, skip];
    }
    
    const result = dfs(root);
    return Math.max(result[0], result[1]);
}

// Enhanced version with visualization
function robWithSteps(nums) {
    const steps = [];
    
    if (nums.length === 0) {
        steps.push("No houses to rob");
        return { maxAmount: 0, steps };
    }
    
    if (nums.length === 1) {
        steps.push(`Only one house: rob it for ${nums[0]}`);
        return { maxAmount: nums[0], steps };
    }
    
    let prev1 = 0; // max up to previous house
    let prev2 = 0; // max up to two houses back
    
    steps.push(`Houses: [${nums}]`);
    steps.push(`Initial: prev1 = ${prev1}, prev2 = ${prev2}`);
    
    for (let i = 0; i < nums.length; i++) {
        const current = Math.max(prev1, nums[i] + prev2);
        steps.push(`\nHouse ${i} (value ${nums[i]}):`);
        steps.push(`  Option 1: Skip -> take prev1 = ${prev1}`);
        steps.push(`  Option 2: Rob -> ${nums[i]} + prev2 = ${nums[i]} + ${prev2} = ${nums[i] + prev2}`);
        steps.push(`  Current max: ${current}`);
        
        prev2 = prev1;
        prev1 = current;
        steps.push(`  Updated: prev1 = ${prev1}, prev2 = ${prev2}`);
    }
    
    steps.push(`\nMaximum amount: ${prev1}`);
    
    return { maxAmount: prev1, steps };
}

console.log(withSteps.steps);
```

**Time Complexity:** O(n) for all DP approaches  
**Space Complexity:** O(n) for tabulation, O(1) for optimized
