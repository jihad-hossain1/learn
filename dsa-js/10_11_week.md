## 1. Maximum Depth of Binary Tree

### Problem Understanding:
Find the maximum depth (height) of a binary tree - the number of nodes along the longest path from root to farthest leaf.

**Example:**
```
    3
   / \
  9  20
     / \
    15  7
Depth: 3
```

### Solutions:

```javascript
class TreeNode {
    constructor(val = 0, left = null, right = null) {
        this.val = val;
        this.left = left;
        this.right = right;
    }
}

// Method 1: Recursive DFS (Optimal)
function maxDepth(root) {
    if (!root) return 0;
    
    const leftDepth = maxDepth(root.left);
    const rightDepth = maxDepth(root.right);
    
    return Math.max(leftDepth, rightDepth) + 1;
}

// Method 2: Iterative BFS (Level Order)
function maxDepthBFS(root) {
    if (!root) return 0;
    
    const queue = [root];
    let depth = 0;
    
    while (queue.length > 0) {
        const levelSize = queue.length;
        
        // Process all nodes at current level
        for (let i = 0; i < levelSize; i++) {
            const node = queue.shift();
            
            if (node.left) queue.push(node.left);
            if (node.right) queue.push(node.right);
        }
        
        depth++;
    }
    
    return depth;
}

// Method 3: Iterative DFS with Stack
function maxDepthDFS(root) {
    if (!root) return 0;
    
    const stack = [{ node: root, depth: 1 }];
    let maxDepth = 0;
    
    while (stack.length > 0) {
        const { node, depth } = stack.pop();
        maxDepth = Math.max(maxDepth, depth);
        
        if (node.right) stack.push({ node: node.right, depth: depth + 1 });
        if (node.left) stack.push({ node: node.left, depth: depth + 1 });
    }
    
    return maxDepth;
}

// Method 4: Tail Recursive (Optimized for some languages)
function maxDepthTailRecursive(root, depth = 0) {
    if (!root) return depth;
    
    return Math.max(
        maxDepthTailRecursive(root.left, depth + 1),
        maxDepthTailRecursive(root.right, depth + 1)
    );
}

// Enhanced version with visualization
function maxDepthWithSteps(root) {
    const steps = [];
    
    function dfs(node, depth = 0) {
        if (!node) {
            steps.push(`Depth ${depth}: null`);
            return 0;
        }
        
        steps.push(`Depth ${depth}: Node ${node.val}`);
        
        const leftDepth = dfs(node.left, depth + 1);
        const rightDepth = dfs(node.right, depth + 1);
        const currentMax = Math.max(leftDepth, rightDepth) + 1;
        
        steps.push(`At node ${node.val}: left=${leftDepth}, right=${rightDepth}, max=${currentMax}`);
        
        return currentMax;
    }
    
    const result = dfs(root);
    return { depth: result, steps };
}

// Helper function to create tree from array
function createTree(arr) {
    if (arr.length === 0) return null;
    
    const root = new TreeNode(arr[0]);
    const queue = [root];
    let i = 1;
    
    while (i < arr.length) {
        const current = queue.shift();
        
        if (arr[i] !== null) {
            current.left = new TreeNode(arr[i]);
            queue.push(current.left);
        }
        i++;
        
        if (i < arr.length && arr[i] !== null) {
            current.right = new TreeNode(arr[i]);
            queue.push(current.right);
        }
        i++;
    }
    
    return root;
}

// Test cases
console.log("Maximum Depth Tests:");
const tree1 = createTree([3, 9, 20, null, null, 15, 7]);
console.log(maxDepth(tree1)); // 3

const tree2 = createTree([1, null, 2]);
console.log(maxDepth(tree2)); // 2

const tree3 = createTree([]);
console.log(maxDepth(tree3)); // 0

// Compare methods
const testTree = createTree([1, 2, 3, 4, 5, 6, 7]);
console.log("Recursive:", maxDepth(testTree));
console.log("BFS:", maxDepthBFS(testTree));
console.log("DFS:", maxDepthDFS(testTree));
```

**Time Complexity:** O(n) - visit each node once  
**Space Complexity:** O(h) for recursion, O(n) for worst-case BFS

## 2. Invert Binary Tree

### Problem Understanding:
Invert a binary tree (mirror it) - swap all left and right children.

**Example:**
```
Input:
     4
   /   \
  2     7
 / \   / \
1   3 6   9

Output:
     4
   /   \
  7     2
 / \   / \
9   6 3   1
```

### Solutions:

```javascript
// Method 1: Recursive (Most Elegant)
function invertTree(root) {
    if (!root) return null;
    
    // Swap left and right subtrees
    [root.left, root.right] = [root.right, root.left];
    
    // Recursively invert subtrees
    invertTree(root.left);
    invertTree(root.right);
    
    return root;
}

// Method 2: Recursive with Return (Functional Style)
function invertTreeFunctional(root) {
    if (!root) return null;
    
    return new TreeNode(
        root.val,
        invertTreeFunctional(root.right), // left becomes inverted right
        invertTreeFunctional(root.left)   // right becomes inverted left
    );
}

// Method 3: Iterative BFS
function invertTreeBFS(root) {
    if (!root) return null;
    
    const queue = [root];
    
    while (queue.length > 0) {
        const node = queue.shift();
        
        // Swap left and right children
        [node.left, node.right] = [node.right, node.left];
        
        if (node.left) queue.push(node.left);
        if (node.right) queue.push(node.right);
    }
    
    return root;
}

// Method 4: Iterative DFS
function invertTreeDFS(root) {
    if (!root) return null;
    
    const stack = [root];
    
    while (stack.length > 0) {
        const node = stack.pop();
        
        // Swap left and right children
        [node.left, node.right] = [node.right, node.left];
        
        if (node.right) stack.push(node.right);
        if (node.left) stack.push(node.left);
    }
    
    return root;
}

// Enhanced version with visualization
function invertTreeWithSteps(root) {
    const steps = [];
    
    function invertWithTracking(node, depth = 0) {
        if (!node) {
            steps.push(`${'  '.repeat(depth)}null`);
            return null;
        }
        
        steps.push(`${'  '.repeat(depth)}Inverting node ${node.val}`);
        steps.push(`${'  '.repeat(depth)}Before: left=${node.left?.val}, right=${node.right?.val}`);
        
        // Swap children
        [node.left, node.right] = [node.right, node.left];
        
        steps.push(`${'  '.repeat(depth)}After: left=${node.left?.val}, right=${node.right?.val}`);
        
        invertWithTracking(node.left, depth + 1);
        invertWithTracking(node.right, depth + 1);
        
        return node;
    }
    
    const result = invertWithTracking(root);
    return { root: result, steps };
}

// Helper to print tree (in-order traversal)
function printTree(root, prefix = '', isLeft = true) {
    if (!root) return;
    
    console.log(prefix + (isLeft ? '└── ' : '├── ') + root.val);
    printTree(root.left, prefix + (isLeft ? '    ' : '│   '), true);
    printTree(root.right, prefix + (isLeft ? '    ' : '│   '), false);
}

// Test cases
console.log("\nInvert Binary Tree Tests:");
const tree = createTree([4, 2, 7, 1, 3, 6, 9]);
console.log("Original tree:");
printTree(tree);

const inverted = invertTree(createTree([4, 2, 7, 1, 3, 6, 9]));
console.log("\nInverted tree:");
printTree(inverted);

// Test with steps
const withSteps = invertTreeWithSteps(createTree([1, 2, 3]));
console.log("\nInversion steps:");
console.log(withSteps.steps.slice(0, 5)); // Show first 5 steps
```

**Time Complexity:** O(n) - visit each node once  
**Space Complexity:** O(h) for recursion, O(n) for worst-case

## 3. Level Order Traversal

### Problem Understanding:
Traverse the tree level by level, returning nodes at each level in separate arrays.

**Example:**
```
Input:
    3
   / \
  9  20
     / \
    15  7

Output: [[3], [9,20], [15,7]]
```

### Solutions:

```javascript
// Method 1: BFS with Queue (Standard)
function levelOrder(root) {
    if (!root) return [];
    
    const result = [];
    const queue = [root];
    
    while (queue.length > 0) {
        const levelSize = queue.length;
        const currentLevel = [];
        
        for (let i = 0; i < levelSize; i++) {
            const node = queue.shift();
            currentLevel.push(node.val);
            
            if (node.left) queue.push(node.left);
            if (node.right) queue.push(node.right);
        }
        
        result.push(currentLevel);
    }
    
    return result;
}

// Method 2: BFS with Level Tracking
function levelOrderLevelTracking(root) {
    if (!root) return [];
    
    const result = [];
    const queue = [{ node: root, level: 0 }];
    
    while (queue.length > 0) {
        const { node, level } = queue.shift();
        
        // Create new level array if needed
        if (level === result.length) {
            result.push([]);
        }
        
        result[level].push(node.val);
        
        if (node.left) queue.push({ node: node.left, level: level + 1 });
        if (node.right) queue.push({ node: node.right, level: level + 1 });
    }
    
    return result;
}

// Method 3: DFS with Level Tracking
function levelOrderDFS(root) {
    const result = [];
    
    function dfs(node, level) {
        if (!node) return;
        
        // Create new level array if needed
        if (level === result.length) {
            result.push([]);
        }
        
        result[level].push(node.val);
        
        dfs(node.left, level + 1);
        dfs(node.right, level + 1);
    }
    
    dfs(root, 0);
    return result;
}

// Method 4: Zigzag Level Order (Variation)
function zigzagLevelOrder(root) {
    if (!root) return [];
    
    const result = [];
    const queue = [root];
    let leftToRight = true;
    
    while (queue.length > 0) {
        const levelSize = queue.length;
        const currentLevel = [];
        
        for (let i = 0; i < levelSize; i++) {
            const node = queue.shift();
            
            // Add to level based on direction
            if (leftToRight) {
                currentLevel.push(node.val);
            } else {
                currentLevel.unshift(node.val);
            }
            
            if (node.left) queue.push(node.left);
            if (node.right) queue.push(node.right);
        }
        
        result.push(currentLevel);
        leftToRight = !leftToRight; // Reverse direction
    }
    
    return result;
}

// Enhanced version with visualization
function levelOrderWithSteps(root) {
    const result = [];
    const steps = [];
    
    if (!root) return { result, steps };
    
    const queue = [root];
    let level = 0;
    
    while (queue.length > 0) {
        const levelSize = queue.length;
        const currentLevel = [];
        
        steps.push(`Level ${level}: Processing ${levelSize} nodes`);
        
        for (let i = 0; i < levelSize; i++) {
            const node = queue.shift();
            currentLevel.push(node.val);
            steps.push(`  Node ${node.val}`);
            
            if (node.left) {
                queue.push(node.left);
                steps.push(`    → Added left child: ${node.left.val}`);
            }
            if (node.right) {
                queue.push(node.right);
                steps.push(`    → Added right child: ${node.right.val}`);
            }
        }
        
        result.push(currentLevel);
        steps.push(`Level ${level} complete: [${currentLevel}]`);
        level++;
    }
    
    return { result, steps };
}

// Test cases
console.log("\nLevel Order Traversal Tests:");
const tree = createTree([3, 9, 20, null, null, 15, 7]);
console.log("Level order:", levelOrder(tree)); // [[3], [9,20], [15,7]]
console.log("DFS level order:", levelOrderDFS(tree)); // Same result
console.log("Zigzag order:", zigzagLevelOrder(tree)); // [[3], [20,9], [15,7]]

// Test with steps
const withSteps = levelOrderWithSteps(createTree([1, 2, 3, 4, 5]));
console.log("Level order with steps:");
console.log(withSteps.result);
```

**Time Complexity:** O(n) - visit each node once  
**Space Complexity:** O(n) for queue storage

## 4. Kth Smallest Element in BST

### Problem Understanding:
Find the kth smallest element in a Binary Search Tree (BST).

**Example:**
```
Input: root = [5,3,6,2,4,null,null,1], k = 3
    5
   / \
  3   6
 / \
2   4
/
1
Output: 3 (3rd smallest is 3)
```

### Solutions:

```javascript
// Method 1: In-order Traversal (Recursive)
function kthSmallest(root, k) {
    const result = [];
    
    function inOrder(node) {
        if (!node) return;
        
        inOrder(node.left);
        result.push(node.val);
        inOrder(node.right);
    }
    
    inOrder(root);
    return result[k - 1];
}

// Method 2: In-order with Early Termination
function kthSmallestEarlyTermination(root, k) {
    let count = 0;
    let result = null;
    
    function inOrder(node) {
        if (!node || result !== null) return;
        
        inOrder(node.left);
        
        count++;
        if (count === k) {
            result = node.val;
            return;
        }
        
        inOrder(node.right);
    }
    
    inOrder(root);
    return result;
}

// Method 3: Iterative In-order
function kthSmallestIterative(root, k) {
    const stack = [];
    let current = root;
    let count = 0;
    
    while (current || stack.length > 0) {
        // Go to leftmost node
        while (current) {
            stack.push(current);
            current = current.left;
        }
        
        // Process node
        current = stack.pop();
        count++;
        
        if (count === k) {
            return current.val;
        }
        
        // Move to right subtree
        current = current.right;
    }
    
    return -1; // k is larger than tree size
}

// Method 4: Augmented Tree (if we can modify structure)
class AugmentedTreeNode {
    constructor(val, left = null, right = null) {
        this.val = val;
        this.left = left;
        this.right = right;
        this.leftCount = 0; // Number of nodes in left subtree
    }
}

function kthSmallestAugmented(root, k) {
    function countNodes(node) {
        if (!node) return 0;
        node.leftCount = countNodes(node.left);
        return node.leftCount + 1 + countNodes(node.right);
    }
    
    countNodes(root);
    
    let current = root;
    while (current) {
        if (k === current.leftCount + 1) {
            return current.val;
        } else if (k <= current.leftCount) {
            current = current.left;
        } else {
            k -= current.leftCount + 1;
            current = current.right;
        }
    }
    
    return -1;
}

// Enhanced version with steps
function kthSmallestWithSteps(root, k) {
    const steps = [];
    let count = 0;
    let result = null;
    
    function inOrder(node, depth = 0) {
        if (!node || result !== null) return;
        
        const indent = '  '.repeat(depth);
        steps.push(`${indent}Visiting node ${node.val}`);
        
        steps.push(`${indent}→ Going left from ${node.val}`);
        inOrder(node.left, depth + 1);
        
        count++;
        steps.push(`${indent}Processing ${node.val} - count = ${count}`);
        
        if (count === k) {
            result = node.val;
            steps.push(`${indent}✓ Found kth smallest: ${node.val}`);
            return;
        }
        
        steps.push(`${indent}→ Going right from ${node.val}`);
        inOrder(node.right, depth + 1);
    }
    
    inOrder(root);
    return { result, steps };
}

// Test cases
console.log("\nKth Smallest in BST Tests:");
const bst = createTree([5, 3, 6, 2, 4, null, null, 1]);
console.log("3rd smallest:", kthSmallest(bst, 3)); // 3
console.log("1st smallest:", kthSmallest(bst, 1)); // 1
console.log("4th smallest:", kthSmallest(bst, 4)); // 4

console.log("Iterative 3rd smallest:", kthSmallestIterative(bst, 3)); // 3

// Test with steps
const withSteps = kthSmallestWithSteps(createTree([4, 2, 6, 1, 3, 5, 7]), 3);
console.log("Kth smallest with steps:");
console.log("Result:", withSteps.result);
console.log("First 10 steps:", withSteps.steps.slice(0, 10));
```

**Time Complexity:** O(h + k) where h is tree height  
**Space Complexity:** O(h) for recursion stack
