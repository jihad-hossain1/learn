## 1. Merge Intervals

### Problem Understanding:
Given an array of intervals, merge all overlapping intervals.

**Example:**
```
Input: [[1,3],[2,6],[8,10],[15,18]]
Output: [[1,6],[8,10],[15,18]]
Explanation: Intervals [1,3] and [2,6] overlap, so they are merged into [1,6].
```

### Solutions:

```javascript
// Method 1: Sort and Merge - Time: O(n log n), Space: O(n)
function merge(intervals) {
    if (intervals.length <= 1) return intervals;
    
    // Sort intervals by start time
    intervals.sort((a, b) => a[0] - b[0]);
    
    const merged = [];
    let currentInterval = intervals[0];
    merged.push(currentInterval);
    
    for (let i = 1; i < intervals.length; i++) {
        const [currentStart, currentEnd] = currentInterval;
        const [nextStart, nextEnd] = intervals[i];
        
        if (currentEnd >= nextStart) {
            // Overlapping intervals, merge them
            currentInterval[1] = Math.max(currentEnd, nextEnd);
        } else {
            // Non-overlapping interval, add to result
            currentInterval = intervals[i];
            merged.push(currentInterval);
        }
    }
    
    return merged;
}

// Method 2: Using While Loop
function mergeWhile(intervals) {
    if (intervals.length <= 1) return intervals;
    
    intervals.sort((a, b) => a[0] - b[0]);
    const result = [intervals[0]];
    
    let i = 1;
    while (i < intervals.length) {
        const last = result[result.length - 1];
        const current = intervals[i];
        
        if (last[1] >= current[0]) {
            // Merge intervals
            last[1] = Math.max(last[1], current[1]);
        } else {
            result.push(current);
        }
        i++;
    }
    
    return result;
}

// Method 3: Functional Approach
function mergeFunctional(intervals) {
    return intervals
        .sort((a, b) => a[0] - b[0])
        .reduce((merged, interval) => {
            const last = merged[merged.length - 1];
            
            if (last && last[1] >= interval[0]) {
                last[1] = Math.max(last[1], interval[1]);
            } else {
                merged.push(interval);
            }
            
            return merged;
        }, []);
}

// Method 4: Interval Tree (Advanced - for frequent merges)
class IntervalTree {
    constructor() {
        this.intervals = [];
    }
    
    addInterval(interval) {
        this.intervals.push(interval);
        this.intervals.sort((a, b) => a[0] - b[0]);
        return this.mergeAll();
    }
    
    mergeAll() {
        return merge(this.intervals);
    }
}

// Enhanced version with visualization
function mergeWithSteps(intervals) {
    const steps = [];
    
    if (intervals.length <= 1) {
        steps.push(`Only ${intervals.length} interval, no merging needed`);
        return { merged: intervals, steps };
    }
    
    steps.push(`Original intervals: ${JSON.stringify(intervals)}`);
    
    // Sort intervals
    intervals.sort((a, b) => a[0] - b[0]);
    steps.push(`Sorted intervals: ${JSON.stringify(intervals)}`);
    
    const merged = [intervals[0]];
    steps.push(`Start with first interval: ${JSON.stringify(intervals[0])}`);
    
    for (let i = 1; i < intervals.length; i++) {
        const last = merged[merged.length - 1];
        const current = intervals[i];
        
        steps.push(`\nComparing: last=${JSON.stringify(last)} vs current=${JSON.stringify(current)}`);
        
        if (last[1] >= current[0]) {
            // Overlap found
            const newEnd = Math.max(last[1], current[1]);
            steps.push(`  Overlap detected! Merging to [${last[0]}, ${newEnd}]`);
            last[1] = newEnd;

            steps.push(`  No overlap, adding new interval: ${JSON.stringify(current)}`);
            merged.push(current);
        }
        
        steps.push(`  Current merged: ${JSON.stringify(merged)}`);
    }
    
    steps.push(`\nFinal merged intervals: ${JSON.stringify(merged)}`);
    return { merged, steps };
}
```

## 2. Search in Rotated Sorted Array

### Problem Understanding:
Search for a target value in a rotated sorted array. Return its index or -1 if not found.

**Example:**
```
Input: nums = [4,5,6,7,0,1,2], target = 0
Output: 4
```

### Solutions:

```javascript
// Method 1: Binary Search (Optimal) - Time: O(log n), Space: O(1)
function search(nums, target) {
    let left = 0;
    let right = nums.length - 1;
    
    while (left <= right) {
        const mid = Math.floor((left + right) / 2);
        
        if (nums[mid] === target) {
            return mid;
        }
        
        // Check which side is sorted
        if (nums[left] <= nums[mid]) {
            // Left side is sorted
            if (nums[left] <= target && target < nums[mid]) {
                // Target is in left sorted portion
                right = mid - 1;
            } else {
                // Target is in right portion
                left = mid + 1;
            }
        } else {
            // Right side is sorted
            if (nums[mid] < target && target <= nums[right]) {
                // Target is in right sorted portion
                left = mid + 1;
            } else {
                // Target is in left portion
                right = mid - 1;
            }
        }
    }
    
    return -1;
}

// Method 2: Find Pivot Then Binary Search
function searchWithPivot(nums, target) {
    // First find the pivot (smallest element)
    function findPivot() {
        let left = 0, right = nums.length - 1;
        
        while (left < right) {
            const mid = Math.floor((left + right) / 2);
            
            if (nums[mid] > nums[right]) {
                left = mid + 1;
            } else {
                right = mid;
            }
        }
        return left;
    }
    
    // Regular binary search
    function binarySearch(left, right, target) {
        while (left <= right) {
            const mid = Math.floor((left + right) / 2);
            
            if (nums[mid] === target) return mid;
            else if (nums[mid] < target) left = mid + 1;
            else right = mid - 1;
        }
        return -1;
    }
    
    const pivot = findPivot();
    
    // Determine which segment to search
    if (target >= nums[pivot] && target <= nums[nums.length - 1]) {
        return binarySearch(pivot, nums.length - 1, target);
    } else {
        return binarySearch(0, pivot - 1, target);
    }
}

// Method 3: One Pass Binary Search (Alternative)
function searchOnePass(nums, target) {
    let left = 0, right = nums.length - 1;
    
    while (left <= right) {
        const mid = Math.floor((left + right) / 2);
        
        if (nums[mid] === target) return mid;
        
        // If left part is sorted
        if (nums[left] <= nums[mid]) {
            if (nums[left] <= target && target < nums[mid]) {
                right = mid - 1;
            } else {
                left = mid + 1;
            }
        } 
        // If right part is sorted
        else {
            if (nums[mid] < target && target <= nums[right]) {
                left = mid + 1;
            } else {
                right = mid - 1;
            }
        }
    }
    
    return -1;
}

// Method 4: Search with Duplicates (Follow-up)
function searchWithDuplicates(nums, target) {
    let left = 0, right = nums.length - 1;
    
    while (left <= right) {
        const mid = Math.floor((left + right) / 2);
        
        if (nums[mid] === target) return true;
        
        // Handle duplicates
        if (nums[left] === nums[mid] && nums[mid] === nums[right]) {
            left++;
            right--;
        }
        // Left side is sorted
        else if (nums[left] <= nums[mid]) {
            if (nums[left] <= target && target < nums[mid]) {
                right = mid - 1;
            } else {
                left = mid + 1;
            }
        }
        // Right side is sorted
        else {
            if (nums[mid] < target && target <= nums[right]) {
                left = mid + 1;
            } else {
                right = mid - 1;
            }
        }
    }
    
    return false;
}

// Enhanced version with visualization
function searchWithSteps(nums, target) {
    const steps = [];
    
    steps.push(`Array: [${nums}]`);
    steps.push(`Target: ${target}`);
    steps.push('');
    
    let left = 0, right = nums.length - 1;
    let step = 1;
    
    while (left <= right) {
        const mid = Math.floor((left + right) / 2);
        
        steps.push(`Step ${step}: left=${left}, right=${right}, mid=${mid}`);
        steps.push(`  nums[mid] = nums[${mid}] = ${nums[mid]}`);
        
        if (nums[mid] === target) {
            steps.push(`  ✓ Found target at index ${mid}!`);
            return { index: mid, steps };
        }
        
        // Check which side is sorted
        if (nums[left] <= nums[mid]) {
            steps.push(`  Left side [${left}..${mid}] is sorted: [${nums.slice(left, mid + 1)}]`);
            
            if (nums[left] <= target && target < nums[mid]) {
                steps.push(`  Target ${target} is in left sorted portion`);
                right = mid - 1;
            } else {
                steps.push(`  Target ${target} is in right portion`);
                left = mid + 1;
            }
        } else {
            steps.push(`  Right side [${mid}..${right}] is sorted: [${nums.slice(mid, right + 1)}]`);
            
            if (nums[mid] < target && target <= nums[right]) {
                steps.push(`  Target ${target} is in right sorted portion`);
                left = mid + 1;
            } else {
                steps.push(`  Target ${target} is in left portion`);
                right = mid - 1;
            }
        }
        
        steps.push(`  New bounds: left=${left}, right=${right}`);
        step++;
    }
    
    steps.push(`✗ Target ${target} not found`);
    return { index: -1, steps };
}
```

## 3. Find Peak Element

### Problem Understanding:
A peak element is an element that is strictly greater than its neighbors. Find any peak element's index.

**Example:**
```
Input: nums = [1,2,3,1]
Output: 2 (element 3 at index 2 is a peak)
```

### Solutions:

```javascript
// Method 1: Linear Scan - Time: O(n), Space: O(1)
function findPeakElementLinear(nums) {
    for (let i = 0; i < nums.length; i++) {
        const left = i === 0 ? -Infinity : nums[i - 1];
        const right = i === nums.length - 1 ? -Infinity : nums[i + 1];
        
        if (nums[i] > left && nums[i] > right) {
            return i;
        }
    }
    return -1;
}

// Method 2: Binary Search (Optimal) - Time: O(log n), Space: O(1)
function findPeakElement(nums) {
    let left = 0;
    let right = nums.length - 1;
    
    while (left < right) {
        const mid = Math.floor((left + right) / 2);
        
        if (nums[mid] > nums[mid + 1]) {
            // Peak is in left half (including mid)
            right = mid;
        } else {
            // Peak is in right half
            left = mid + 1;
        }
    }
    
    return left;
}

// Method 3: Recursive Binary Search
function findPeakElementRecursive(nums) {
    function search(left, right) {
        if (left === right) return left;
        
        const mid = Math.floor((left + right) / 2);
        
        if (nums[mid] > nums[mid + 1]) {
            return search(left, mid);
        } else {
            return search(mid + 1, right);
        }
    }
    
    return search(0, nums.length - 1);
}

// Method 4: Find All Peak Elements
function findAllPeakElements(nums) {
    const peaks = [];
    
    for (let i = 0; i < nums.length; i++) {
        const left = i === 0 ? -Infinity : nums[i - 1];
        const right = i === nums.length - 1 ? -Infinity : nums[i + 1];
        
        if (nums[i] > left && nums[i] > right) {
            peaks.push(i);
        }
    }
    
    return peaks;
}

// Method 5: Find Global Peak (Maximum)
function findGlobalPeak(nums) {
    return nums.indexOf(Math.max(...nums));
}

// Enhanced version with visualization
function findPeakWithSteps(nums) {
    const steps = [];
    
    steps.push(`Array: [${nums}]`);
    steps.push(`Length: ${nums.length}`);
    steps.push('');
    
    let left = 0, right = nums.length - 1;
    let step = 1;
    
    while (left < right) {
        const mid = Math.floor((left + right) / 2);
        
        steps.push(`Step ${step}: left=${left}, right=${right}, mid=${mid}`);
        steps.push(`  nums[${mid}] = ${nums[mid]}, nums[${mid + 1}] = ${nums[mid + 1]}`);
        
        if (nums[mid] > nums[mid + 1]) {
            steps.push(`  nums[${mid}] > nums[${mid + 1}] → Peak is in left half [${left}..${mid}]`);
            right = mid;
        } else {
            steps.push(`  nums[${mid}] <= nums[${mid + 1}] → Peak is in right half [${mid + 1}..${right}]`);
            left = mid + 1;
        }
        
        steps.push(`  New bounds: left=${left}, right=${right}`);
        step++;
    }
    
    steps.push(`\n✓ Peak found at index ${left}: nums[${left}] = ${nums[left]}`);
    
    // Verify it's actually a peak
    const leftNeighbor = left === 0 ? -Infinity : nums[left - 1];
    const rightNeighbor = left === nums.length - 1 ? -Infinity : nums[left + 1];
    
    steps.push(`Verification: ${leftNeighbor} < ${nums[left]} > ${rightNeighbor}`);
    
    return { peakIndex: left, peakValue: nums[left], steps };
}
```

## 4. K Closest Points to Origin

### Problem Understanding:
Given an array of points and integer k, return the k closest points to the origin (0,0).

**Example:**
```
Input: points = [[1,3],[-2,2]], k = 1
Output: [[-2,2]]
Explanation: Distance from origin: [√10, √8] → [-2,2] is closest
```

### Solutions:

```javascript
// Method 1: Sort All Points - Time: O(n log n), Space: O(n)
function kClosestSort(points, k) {
    // Calculate squared distance (avoid sqrt for efficiency)
    function distance(point) {
        return point[0] * point[0] + point[1] * point[1];
    }
    
    return points
        .sort((a, b) => distance(a) - distance(b))
        .slice(0, k);
}

// Method 2: Max Heap - Time: O(n log k), Space: O(k)
class MaxHeap {
    constructor() {
        this.heap = [];
    }
    
    push(point, distance) {
        this.heap.push({ point, distance });
        this._bubbleUp();
        
        if (this.heap.length > k) {
            this.pop();
        }
    }
    
    pop() {
        const max = this.heap[0];
        const end = this.heap.pop();
        
        if (this.heap.length > 0) {
            this.heap[0] = end;
            this._sinkDown();
        }
        
        return max;
    }
    
    toArray() {
        return this.heap.map(item => item.point);
    }
    
    _bubbleUp() {
        let index = this.heap.length - 1;
        const element = this.heap[index];
        
        while (index > 0) {
            const parentIndex = Math.floor((index - 1) / 2);
            const parent = this.heap[parentIndex];
            
            if (element.distance <= parent.distance) break;
            
            this.heap[parentIndex] = element;
            this.heap[index] = parent;
            index = parentIndex;
        }
    }
    
    _sinkDown() {
        let index = 0;
        const length = this.heap.length;
        const element = this.heap[0];
        
        while (true) {
            let leftChildIndex = 2 * index + 1;
            let rightChildIndex = 2 * index + 2;
            let leftChild, rightChild;
            let swap = null;
            
            if (leftChildIndex < length) {
                leftChild = this.heap[leftChildIndex];
                if (leftChild.distance > element.distance) {
                    swap = leftChildIndex;
                }
            }
            
            if (rightChildIndex < length) {
                rightChild = this.heap[rightChildIndex];
                if ((swap === null && rightChild.distance > element.distance) ||
                    (swap !== null && rightChild.distance > leftChild.distance)) {
                    swap = rightChildIndex;
                }
            }
            
            if (swap === null) break;
            
            this.heap[index] = this.heap[swap];
            this.heap[swap] = element;
            index = swap;
        }
    }
}

function kClosestHeap(points, k) {
    const heap = new MaxHeap();
    
    for (let point of points) {
        const distance = point[0] * point[0] + point[1] * point[1];
        heap.push(point, distance);
    }
    
    return heap.toArray();
}

// Method 3: QuickSelect - Time: O(n) average, Space: O(n)
function kClosestQuickSelect(points, k) {
    function distance(point) {
        return point[0] * point[0] + point[1] * point[1];
    }
    
    function quickSelect(left, right, k) {
        if (left >= right) return;
        
        const pivotIndex = partition(left, right);
        
        if (pivotIndex === k) {
            return;
        } else if (pivotIndex < k) {
            quickSelect(pivotIndex + 1, right, k);
        } else {
            quickSelect(left, pivotIndex - 1, k);
        }
    }
    
    function partition(left, right) {
        const pivotDistance = distance(points[right]);
        let i = left;
        
        for (let j = left; j < right; j++) {
            if (distance(points[j]) <= pivotDistance) {
                [points[i], points[j]] = [points[j], points[i]];
                i++;
            }
        }
        
        [points[i], points[right]] = [points[right], points[i]];
        return i;
    }
    
    quickSelect(0, points.length - 1, k);
    return points.slice(0, k);
}

// Method 4: Using Built-in Priority Queue (if available)
function kClosestBuiltIn(points, k) {
    const distances = points.map((point, index) => ({
        point,
        distance: point[0] * point[0] + point[1] * point[1],
        index
    }));
    
    distances.sort((a, b) => a.distance - b.distance);
    return distances.slice(0, k).map(item => item.point);
}

// Enhanced version with visualization
function kClosestWithSteps(points, k) {
    const steps = [];
    
    steps.push(`Points: ${JSON.stringify(points)}`);
    steps.push(`k: ${k}`);
    steps.push('');
    
    // Calculate distances
    const pointsWithDistance = points.map((point, index) => {
        const distance = point[0] * point[0] + point[1] * point[1];
        return { point, distance, index };
    });
    
    steps.push("Calculated distances:");
    pointsWithDistance.forEach(p => {
        steps.push(`  Point ${p.index}: [${p.point}] → distance² = ${p.distance}`);
    });
    
    // Sort by distance
    pointsWithDistance.sort((a, b) => a.distance - b.distance);
    
    steps.push(`\nSorted by distance:`);
    pointsWithDistance.forEach((p, i) => {
        steps.push(`  ${i + 1}. [${p.point}] → ${p.distance}`);
    });
    
    const result = pointsWithDistance.slice(0, k).map(p => p.point);
    
    steps.push(`\nTop ${k} closest points: ${JSON.stringify(result)}`);
    
    return { closest: result, steps };
}
```
